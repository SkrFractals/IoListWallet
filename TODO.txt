bool AutoInputs(List<Txo> inputs, List<Txo> outputs, List<Txo> utxos, out List<Txo> Add){
var s = MyNetwork.GetValue();
foreach(var o in outputs) s.Add(o.GetValue());
List<Txo> _u = []
foreach(var u in utxos)
	_u.Add(u);
void R(Txo i){
s.Subtract(i);
_u.Remove(i);
}
foreach(var i in inputs) R(i);

while(!s.IsEmpty()) { // As long as at least something is missing in inputs;
	Txo? best = null;
	double bestFill = 0, bestOverfill = double.MaxValue;
	foreach(var u in _u){
		var (fill, overfill) = u.Fills(s);
		if(fill > bestFill || (fill >= bestFill && overfill < bestOverfill)){
			bestFill = fill; bestOverfill = overfill; Add.Add(best = u);
		}
	}
	if(best == null)
		return false;
 	R(best);
}
return true;
}

bool AutoInputs(IoTransaction T, IoAccount A, out List<Txo> Add) => AutoInputs(T.Inputs, T.Outputs, A.Utxos,  out Add);
